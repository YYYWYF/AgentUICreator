export function ConversationComposerPlugin(props: ComponentProps<typeof Composer>) {
  const { rootProps, ...rest } = props;
  return (
    <Composer
      {...rest}
      rootProps={{
        ...rootProps,
        style: {
          ...rootProps?.style,
          padding: 12,
