编辑 1 — 新建 src/components/AgentUIBridge.vue：

<template><div ref="container"></div></template>
<script>

export default {
  name: 'AgentUIBridge',
  props: { endpoint: { type: String, default: "/agent" }, locale: { type: String, default: "en-US" }, theme: { type: String, default: "violet" }, threadId: String },
  emits: ['agent-ready', 'thread-change', 'agent-error'],
  async mounted() {
    const moduleUrl = new URL("/agent-ui.js", window.location.href).href;
    try { await import(/* @vite-ignore */ moduleUrl); }
    catch (error) { if (!this._agentDisposed) this.$emit('agent-error', { code: 'AGENT_UI_MODULE_LOAD_ERROR', error }); return; }
    if (this._agentDisposed) return;
    const element = document.createElement('agent-ui');
    this._agentElement = element;
    element.config = { ...{"endpoint":"/agent","locale":"en-US","theme":"violet"}, endpoint: this.endpoint, locale: this.locale, theme: this.theme, threadId: this.threadId };
    this._agentListeners = ['agent-ready', 'thread-change', 'agent-error'].map(name => {
      const listener = event => this.$emit(name, event.detail);
      element.addEventListener(name, listener);
      return [name, listener];
    });
    this.$refs.container.append(element);
  },
  watch: {
    endpoint: 'updateAgentConfig', locale: 'updateAgentConfig', theme: 'updateAgentConfig', threadId: 'updateAgentConfig',
  },
  methods: {
    updateAgentConfig() { if (this._agentElement) this._agentElement.config = { endpoint: this.endpoint, locale: this.locale, theme: this.theme, threadId: this.threadId }; },
    disposeAgentElement() {
      this._agentDisposed = true;
      if (!this._agentElement) return;
      for (const [name, listener] of this._agentListeners) this._agentElement.removeEventListener(name, listener);
      this._agentElement.remove();
      this._agentElement = undefined;
    },
  },
  beforeUnmount() { this.disposeAgentElement(); },
};
</script>
