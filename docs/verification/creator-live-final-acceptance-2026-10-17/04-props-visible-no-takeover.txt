import { ConversationComposerPlugin as Composer } from "@za38.03/agent-ui-plugins/composer";
import type { ComponentProps } from "react";

export function ConversationComposerPlugin(props: ComponentProps<typeof Composer>) {
  return (
    <Composer
      {...props}
      rootProps={{
        ...props.rootProps,
        style: {
          ...props.rootProps?.style,
          padding: 12,
          borderRadius: 20,
          backgroundColor: "#f3e8ff",
        },
      }}
    />
  );
}


要点：

通过公开 prop rootProps.style 设置 padding: 12、borderRadius: 20、backgroundColor: '#f3e8ff'，并合并/保留了上游可能传入的 rootProps 与其原有 style。
其余 props 全部通过 {...props} 透传；未改动 AppUIModel、未改 npm 包实现、未接管源码。
